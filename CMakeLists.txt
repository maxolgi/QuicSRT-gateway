cmake_minimum_required(VERSION 3.24)

project(websrt-gateway
    VERSION 0.1.0
    DESCRIPTION "CLI-parity C++ replacement of the Rust WebSRT gateway"
    LANGUAGES CXX C)

# Strict warnings for OUR targets only; deps/libwebsrt and everything it
# pulls in (srt fork, lsquic) build with their own settings.
add_library(gateway_warnings INTERFACE)
target_compile_options(gateway_warnings INTERFACE -Wall -Wextra -Wpedantic)

# The gateway is a libwebsrt CONSUMER: it sees the library exclusively
# through the exported interface targets (websrt::websrt, websrt::srt-deps,
# websrt::wt-deps) - never raw deps/ include paths. websrt::wt-deps also
# propagates LSQUIC_WEBTRANSPORT_SERVER_SUPPORT, the define lsquic.h gates
# the server-side WebTransport API behind. BoringSSL must already
# be fetched + built inside the libwebsrt checkout (./build.sh setup).
# The gateway is the only thing this project installs. The dependency
# chain pulled in below (srt fork, lsquic) carries its own install()
# rules; shadow the builtin with a no-op for the duration of its
# add_subdirectory. Shadowing moves the builtin to _install, so this
# project's own rules (further down) call it directly - cmake --install
# then ships exactly the binary + the two conf files (P4.4).
set(WEBSRT_ENABLE_WT ON CACHE BOOL "WebTransport support (the gateway requires it)")
function(install)
endfunction()
add_subdirectory(deps/libwebsrt)

if(NOT WEBSRT_ENABLE_WT)
    message(FATAL_ERROR "websrt-gateway requires WEBSRT_ENABLE_WT=ON")
endif()

find_package(Threads REQUIRED)

# ---- The embedded player (rust-embed parity) ----
# web/dist is fetched from the latest WebSRT release (./build.sh setup or
# ./build.sh web-dist) and baked into the binary: a generated table in
# CMAKE_CURRENT_BINARY_DIR. Generation runs at CONFIGURE time; the glob is
# CONFIGURE_DEPENDS, so refreshing web/dist + re-running the configure
# step (build.sh web-dist does both) regenerates and rebuilds the table.
set(WEB_DIST_DIR ${CMAKE_CURRENT_SOURCE_DIR}/web/dist)
if(NOT EXISTS ${WEB_DIST_DIR}/index.html)
    message(FATAL_ERROR
        "web/dist is missing (the embedded player): run ./build.sh setup "
        "or ./build.sh web-dist to fetch it from the WebSRT release")
endif()
file(GLOB_RECURSE WEB_DIST_FILES LIST_DIRECTORIES false RELATIVE ${WEB_DIST_DIR}
     CONFIGURE_DEPENDS ${WEB_DIST_DIR}/*)
list(SORT WEB_DIST_FILES)
set(EMBEDDED_PLAYER_SRC ${CMAKE_CURRENT_BINARY_DIR}/embedded_player.cpp)
set(_ep_content "// Generated by CMake from web/dist - do not edit.\n#include \"gateway/embedded_player.h\"\nnamespace websrt::gateway {\nnamespace {\n")
set(_ep_idx 0)
foreach(_f IN LISTS WEB_DIST_FILES)
    file(READ ${WEB_DIST_DIR}/${_f} _ep_hex HEX)
    string(REGEX REPLACE "(..)" "0x\\1," _ep_bytes "${_ep_hex}")
    # 16 bytes per line (5 chars per "0x..," token = 80 columns).
    string(REGEX REPLACE "(.{80})" "\\1\n" _ep_bytes "${_ep_bytes}")
    string(APPEND _ep_content
        "const unsigned char kAsset${_ep_idx}[] = {\n${_ep_bytes}};\n")
    math(EXPR _ep_idx "${_ep_idx} + 1")
endforeach()
string(APPEND _ep_content "} // namespace\n\nconst std::vector<EmbeddedAsset>& EmbeddedPlayerFiles()\n{\n    static const std::vector<EmbeddedAsset> kFiles{\n")
set(_ep_idx 0)
foreach(_f IN LISTS WEB_DIST_FILES)
    string(APPEND _ep_content
        "        EmbeddedAsset{std::string_view(\"/${_f}\"), "
        "std::string_view(reinterpret_cast<const char*>(kAsset${_ep_idx}), sizeof kAsset${_ep_idx})},\n")
    math(EXPR _ep_idx "${_ep_idx} + 1")
endforeach()
string(APPEND _ep_content "    };\n    return kFiles;\n}\n\n} // namespace websrt::gateway\n")
file(WRITE ${EMBEDDED_PLAYER_SRC} "${_ep_content}")
message(STATUS "Embedded player: ${WEB_DIST_DIR} (${_ep_idx} files)")

# The example config, embedded the same way, so a bare binary can write
# ./gateway.conf on first run (self-documenting release artifact).
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/scripts/gateway.example.conf _example_hex HEX)
string(REGEX REPLACE "(..)" "0x\\1," _example_bytes "${_example_hex}")
string(REGEX REPLACE "(.{80})" "\\1\n" _example_bytes "${_example_bytes}")
set(EMBEDDED_EXAMPLE_SRC ${CMAKE_CURRENT_BINARY_DIR}/embedded_example_conf.cpp)
file(WRITE ${EMBEDDED_EXAMPLE_SRC}
    "// Generated by CMake from scripts/gateway.example.conf - do not edit.\n"
    "#include \"gateway/embedded_example_conf.h\"\n"
    "namespace websrt::gateway {\nnamespace {\n"
    "const unsigned char kExampleConf[] = {\n${_example_bytes}};\n"
    "} // namespace\n\n"
    "std::string_view EmbeddedExampleConf()\n{\n"
    "    return {reinterpret_cast<const char*>(kExampleConf), sizeof kExampleConf};\n"
    "}\n\n} // namespace websrt::gateway\n")

# Gateway core library (refinement item 1): everything the embeddable
# gateway owns - accept loop machinery, session policy seam, registry - so a
# C++ app can link the gateway in-process. The CLI binary only wires
# CLI/config/ingest. Include dir exposes "gateway/..." headers.
add_library(websrtgw STATIC
    ${EMBEDDED_EXAMPLE_SRC}
    ${EMBEDDED_PLAYER_SRC}
    src/gateway/cert.cpp
    src/gateway/gateway.cpp
    src/gateway/policy.cpp
    src/gateway/selftest.cpp
    src/gateway/session_registry.cpp
    src/gateway/srt_initiator.cpp
    src/gateway/ingest/srt_ingest.cpp
    src/gateway/web_server.cpp
    src/gateway/webhooks.cpp
    src/gateway/wt_server.cpp)
target_compile_features(websrtgw PUBLIC cxx_std_20)
set_target_properties(websrtgw PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(websrtgw PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(websrtgw PUBLIC gateway_warnings Threads::Threads websrt::websrt websrt::wt-deps)

# Thin CLI binary over the core library.
#
# toml++ (header-only, deps/tomlplusplus) is a gateway-binary-side
# dependency by design (refinement item 2): config parsing lives in
# websrtgw_config, never in the embeddable websrtgw core.
add_library(tomlplusplus INTERFACE)
target_include_directories(tomlplusplus INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/deps/tomlplusplus/include)

add_library(websrtgw_config STATIC
    src/gateway/config.cpp
    src/gateway/reload.cpp)
target_compile_features(websrtgw_config PUBLIC cxx_std_20)
set_target_properties(websrtgw_config PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(websrtgw_config PUBLIC gateway_warnings websrtgw tomlplusplus)

add_executable(websrt-gateway src/main.cpp)
target_link_libraries(websrt-gateway PRIVATE websrtgw_config)

# P4.4: cmake --install ships the binary, the example config and the
# supervisord conf (GNUInstallDirs destinations; no CPack, no components).
# With the default prefix (/usr/local) the binary lands exactly where
# scripts/websrt-cpp.conf's command= expects it; the conf files land under
# <prefix>/etc and are copied to their runtime homes at deployment (README,
# "Deployment"). _install IS the install builtin - see the shadow above.
include(GNUInstallDirs)
_install(TARGETS websrt-gateway RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
_install(FILES scripts/gateway.example.conf DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/websrt-cpp)
_install(FILES scripts/websrt-cpp.conf DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/supervisor/conf.d)
